cmake_minimum_required(VERSION 3.16)
project(catchthecat-wasm CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/get_cpm.cmake)
CPMAddPackage(NAME glm GITHUB_REPOSITORY g-truc/glm GIT_TAG 1.0.1)
file(GLOB CAT_BOT_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/apps/catchthecat/*.cpp)
list(FILTER CAT_BOT_SRC EXCLUDE REGEX "main\\.cpp$")
add_executable(${CAT_BOT_NAME} ${CAT_BOT_SRC} ${CMAKE_CURRENT_SOURCE_DIR}/wasm_main.cpp)
target_include_directories(${CAT_BOT_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/apps/catchthecat ${CMAKE_CURRENT_SOURCE_DIR}/core/sources)
target_link_libraries(${CAT_BOT_NAME} PRIVATE glm::glm)
set_target_properties(${CAT_BOT_NAME} PROPERTIES SUFFIX ".js")
target_link_options(${CAT_BOT_NAME} PRIVATE
    -sMODULARIZE=1
    -sEXPORT_ES6=1
    -sEXPORT_NAME=CatBot
    -sINVOKE_RUN=0
    -sEXIT_RUNTIME=0
    "-sEXPORTED_RUNTIME_METHODS=['callMain']"
    -sALLOW_MEMORY_GROWTH=1
    -sSTACK_SIZE=8388608
    -sENVIRONMENT=web,worker
)
